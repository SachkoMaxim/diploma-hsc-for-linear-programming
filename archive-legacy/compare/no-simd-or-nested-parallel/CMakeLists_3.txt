cmake_minimum_required(VERSION 3.30)
project(diploma_hsc_for_linear_programming)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenMP REQUIRED)
find_package(TBB REQUIRED)

add_executable(diploma_hsc_for_linear_programming
        main.cpp
        src/SharedData.cpp
        src/SimplexMath.h
        src/ConsoleVisualizer.h
        src/ParallelSimplex.cpp
        src/FileWriter.h
)

target_include_directories(diploma_hsc_for_linear_programming PRIVATE src)

target_link_libraries(diploma_hsc_for_linear_programming PRIVATE
        OpenMP::OpenMP_CXX
        TBB::tbb
)

if(MSVC)
    # /Qvec- ПРИМУСОВО вимикає автовекторизацію в MSVC
    target_compile_options(diploma_hsc_for_linear_programming PRIVATE /O2 /fp:fast /Qvec-)
else()
    # -fno-tree-vectorize ЗАБОРОНЯЄ компіляторам GCC/Clang генерувати SIMD інструкції
    # -Wno-unknown-pragmas приховує попередження, якщо у коді десь залишаться прагми simd
    target_compile_options(diploma_hsc_for_linear_programming PRIVATE -O3 -march=native -ffast-math -fno-tree-vectorize -Wno-unknown-pragmas)
endif()
